library ieee;
use ieee.std_logic_1164.all;

entity tb_modN_display is
end tb_modN_display;

architecture sim of tb_modN_display is
  signal clk  : std_logic := '0';
  signal rst  : std_logic := '1';
  signal SEG  : std_logic_vector(6 downto 0);

begin
  UUT : entity work.modN_display
    generic map (
      N          => 10,
      ACTIVE_LOW => true
    )
    port map (
      clk => clk,
      rst => rst,
      SEG => SEG
    );

  -- horloge 50 ns
  clk_proc : process
  begin
    while true loop
      clk <= '0'; wait for 25 ns;
      clk <= '1'; wait for 25 ns;
    end loop;
  end process;

  stim_proc : process
  begin
    rst <= '1'; wait for 50 ns;
    rst <= '0'; wait;
  end process;

end sim;
